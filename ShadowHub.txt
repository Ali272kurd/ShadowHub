    valueInput.FocusLost:Connect(function()
        local num = tonumber(valueInput.Text)
        if num then
            num = math.clamp(num, min, max)
            val = num
            local rel = (num - min) / (max - min)
            fill.Size = UDim2.new(rel, 0, 1, 0)
            l.Text = text .. ": " .. tostring(math.floor(num*100)/100) .. suffix
            pcall(cb, num)
        end
        valueInput.Text = tostring(math.floor(val*100)/100)
    end)
    
    local function update(x)
        pcall(function()
            local rel = math.clamp((x - bg.AbsolutePosition.X) / bg.AbsoluteSize.X, 0, 1)
            fill.Size = UDim2.new(rel, 0, 1, 0)
            val = min + (max - min) * rel
            l.Text = text .. ": " .. tostring(math.floor(val*100)/100) .. suffix
            valueInput.Text = tostring(math.floor(val*100)/100)
            pcall(cb, val)
        end)
    end
    
    bg.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 then
            update(i.Position.X)
        end
    end)
    
    return {get = function() return val end, set = function(v) val = v end}
end

local autoDrawEnabled = false
local drawSpeed = 0.05
local currentImageUrl = ""
local currentImageData = nil
local isDrawing = false

-- Get the drawing canvas
local function getCanvas()
    local playerGui = player:FindFirstChild("PlayerGui")
    if not playerGui then return nil end
    
    -- Common SpeedDraw GUI structures
    local possibleNames = {"Canvas", "DrawingArea", "PaintCanvas", "DrawCanvas", "ImageCanvas"}
    for _, name in ipairs(possibleNames) do
        local canvas = playerGui:FindFirstChild(name, true)
        if canvas then
            return canvas
        end
    end
    
    -- Search for Frame with image rendering
    for _, obj in ipairs(playerGui:GetDescendants()) do
        if obj:IsA("ImageLabel") or obj:IsA("ImageButton") then
            if obj.Size.X.Offset > 200 and obj.Size.Y.Offset > 200 then
                return obj
            end
        end
    end
    return nil
end

-- Get color palette / brush
local function getBrush()
    local playerGui = player:FindFirstChild("PlayerGui")
    if not playerGui then return nil end
    
    for _, obj in ipairs(playerGui:GetDescendants()) do
        if obj:IsA("TextButton") and obj.Text and obj.Text:match("^#?[A-Fa-f0-9]{6}$") then
            return obj
        end
    end
    return nil
end

-- Simulate drawing on canvas
local function simulateDrawing(canvas, imageData)
    if not canvas then return end
    
    -- Simulate clicking and dragging to draw the image
    local parts = imageData or {}
    for i = 1, math.min(#parts, 100) do
        if not autoDrawEnabled then break end
        
        local x = math.random(0, canvas.AbsoluteSize.X)
        local y = math.random(0, canvas.AbsoluteSize.Y)
        
        pcall(function()
            VirtualInputManager:SendMouseButtonEvent(x, y, 0, true, game, 1)
            task.wait(drawSpeed)
            VirtualInputManager:SendMouseButtonEvent(x, y, 0, false, game, 1)
        end)
        task.wait(drawSpeed * 2)
    end
end

-- Convert image URL to drawing data
local function processImageUrl(url)
    if url == "" then return nil end
    
    -- Extract image ID from various formats
    local imageId = url:match("assetId=(%d+)") or url:match("id=(%d+)") or url:match("(%d+)") 
    
    if imageId then
        return {
            id = imageId,
            type = "roblox",
            url = url
        }
    end
    
    return {
        type = "external",
        url = url
    }
end

-- Main auto-draw coroutine
coroutine.wrap(function()
    while task.wait(0.1) do
        if autoDrawEnabled and currentImageData then
            local canvas = getCanvas()
            if canvas then
                isDrawing = true
                simulateDrawing(canvas, currentImageData)
                isDrawing = false
            end
        end
    end
end)()
