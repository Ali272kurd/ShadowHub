local function parseColor(input)
    if typeof(input) == "Color3" then return input end
    local str = string.lower(tostring(input or "purple"))
    
    if str == "rainbow" then
        local hue = (rainbowTick % 5) / 5
        return Color3.fromHSV(hue, 1, 1)
    end
    
    local presets = {
        red = Color3.fromRGB(255, 60, 60),
        green = Color3.fromRGB(60, 255, 100),
        blue = Color3.fromRGB(60, 120, 255),
        purple = Color3.fromRGB(180, 80, 255),
        pink = Color3.fromRGB(255, 105, 180),
        yellow = Color3.fromRGB(255, 220, 50),
        orange = Color3.fromRGB(255, 140, 0),
        cyan = Color3.fromRGB(0, 225, 255),
        white = Color3.fromRGB(255, 255, 255),
        black = Color3.fromRGB(15, 15, 15),
        gray = Color3.fromRGB(140, 140, 140),
        lime = Color3.fromRGB(50, 255, 50),
        crimson = Color3.fromRGB(220, 20, 60),
        indigo = Color3.fromRGB(75, 0, 130),
        teal = Color3.fromRGB(0, 128, 128),
        gold = Color3.fromRGB(255, 215, 0)
    }
    
    if presets[str] then return presets[str] end
    
    if string.find(str, ",") then
        local parts = string.split(str, ",")
        local r, g, b = tonumber(parts[1]), tonumber(parts[2]), tonumber(parts[3])
        if r and g and b then
            return Color3.fromRGB(r, g, b)
        end
    end
    
    return Color3.fromRGB(180, 80, 255)
end

local espDrawings = {}

local function removeEspForPlayer(p)
    if espDrawings[p] then
        if espDrawings[p].box then pcall(function() espDrawings[p].box:Remove() end) end
        if espDrawings[p].name then pcall(function() espDrawings[p].name:Remove() end) end
        if espDrawings[p].details then pcall(function() espDrawings[p].details:Remove() end) end
        if espDrawings[p].trace then pcall(function() espDrawings[p].trace:Remove() end) end
        espDrawings[p] = nil
    end
end

Players.PlayerRemoving:Connect(function(p)
    removeEspForPlayer(p)
end)

local FOVCircle = Drawing.new("Circle")
FOVCircle.Visible = false
FOVCircle.Thickness = 1.5
FOVCircle.Color = parseColor(config.themeColorName)
FOVCircle.Filled = false

local function getClosestPlayer()
    local closest = nil
    local shortestDist = config.fovRadius
    local cam = Workspace.CurrentCamera
    local screenCenter = Vector2.new(cam.ViewportSize.X / 2, cam.ViewportSize.Y / 2)
    
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= player and p.Character and p.Character:FindFirstChild(config.targetPart) then
            local hum = p.Character:FindFirstChildOfClass("Humanoid")
            if hum and hum.Health > 0 then
                local targetPart = p.Character[config.targetPart]
                local screenPoint, onScreen = cam:WorldToViewportPoint(targetPart.Position)
                if onScreen then
                    local dist = (screenCenter - Vector2.new(screenPoint.X, screenPoint.Y)).Magnitude
                    if dist < shortestDist then
                        shortestDist = dist
                        closest = p
                    end
                end
            end
        end
    end
    return closest
end

local lockedPlayer = nil

local function toggleLock()
    if lockedPlayer then
        lockedPlayer = nil
        camlockBtn.TextColor3 = parseColor(config.themeColorName)
        infoFrame.Visible = false
    else
        lockedPlayer = getClosestPlayer()
        if lockedPlayer then
            camlockBtn.TextColor3 = Color3.fromRGB(0, 255, 120)
            targetNameLbl.Text = lockedPlayer.DisplayName
            targetAvatarImg.Image = "rbxthumb://type=AvatarHeadShot&id=" .. lockedPlayer.UserId .. "&w=150&h=150"
            infoFrame.Visible = true
        end
    end
end

camlockBtn.MouseButton1Click:Connect(toggleLock)

UserInputService.InputBegan:Connect(function(input, gp)
    if gp then return end
    if input.KeyCode == config.keybind then
        toggleLock()
    end
end)

RunService.RenderStepped:Connect(function()
    local cam = Workspace.CurrentCamera
    local viewportCenter = Vector2.new(cam.ViewportSize.X / 2, cam.ViewportSize.Y / 2)
    
    camlockBtn.Visible = config.enabled
    
    FOVCircle.Visible = config.showFov and config.enabled
    FOVCircle.Radius = config.fovRadius
    FOVCircle.Position = viewportCenter
    FOVCircle.Color = parseColor(config.themeColorName)
    
    menuToggleStroke.Color = parseColor(config.themeColorName)
    menuToggleBtn.TextColor3 = parseColor(config.themeColorName)
    camlockBtnStroke.Color = parseColor(config.themeColorName)
    infoStroke.Color = parseColor(config.themeColorName)
    
    if player.Character and player.Character:FindFirstChildOfClass("Humanoid") then
        local hum = player.Character:FindFirstChildOfClass("Humanoid")
        if config.speedEnabled then
            hum.WalkSpeed = config.speedValue
        else
            hum.WalkSpeed = 16
        end
        
        if config.jumpEnabled then
            hum.JumpPower = config.jumpValue
            hum.UseJumpPower = true
        else
            hum.JumpPower = 50
        end
    end
    
    if config.enabled and lockedPlayer and lockedPlayer.Character then
        local targetPartObj = lockedPlayer.Character:FindFirstChild(config.targetPart)
        local hum = lockedPlayer.Character:FindFirstChildOfClass("Humanoid")
        
        if targetPartObj and hum and hum.Health > 0 then
            local targetPos = targetPartObj.Position + (targetPartObj.AssemblyLinearVelocity * config.prediction)
            cam.CFrame = cam.CFrame:Lerp(CFrame.new(cam.CFrame.Position, targetPos), config.smoothness)
            
            local distanceMeters = math.round((targetPartObj.Position - cam.CFrame.Position).Magnitude)
            local currentHealth = math.round(hum.Health)
            targetDetailsLbl.Text = "HP: " .. currentHealth .. " | dist: " .. distanceMeters .. "m"
        else
            lockedPlayer = nil
            camlockBtn.TextColor3 = parseColor(config.themeColorName)
            infoFrame.Visible = false
        end
    else
        if not lockedPlayer then
            infoFrame.Visible = false
        end
    end
    
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= player then
            local char = p.Character
            local hum = char and char:FindFirstChildOfClass("Humanoid")
            local rootPart = char and char:FindFirstChild("HumanoidRootPart")
            
            if config.espEnabled and char and hum and rootPart and hum.Health > 0 then
                local pos, onScreen = cam:WorldToViewportPoint(rootPart.Position)
                if onScreen then
                    if not espDrawings[p] then
                        espDrawings[p] = {
                            box = Drawing.new("Square"),
                            name = Drawing.new("Text"),
                            details = Drawing.new("Text"),
                            trace = Drawing.new("Line")
                        }
                        espDrawings[p].box.Filled = false
                        espDrawings[p].box.Thickness = 1.5
